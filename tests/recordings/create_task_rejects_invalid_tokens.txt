oidc with malformed email -> InvalidArgument: Request contains an invalid argument.
oidc with audience without scheme -> Some(
    HttpRequest(
        HttpRequest {
            url: "https://example.com/work",
            http_method: Post,
            headers: {
                "User-Agent": "Google-Cloud-Tasks",
            },
            body: [],
            authorization_header: Some(
                OidcToken(
                    OidcToken {
                        service_account_email: "{service_account}",
                        audience: "audience",
                    },
                ),
            ),
        },
    ),
)

oauth without email -> InvalidArgument: service_account_email must be set.
oauth with malformed email -> InvalidArgument: Request contains an invalid argument.
oauth with scope without scheme -> Some(
    HttpRequest(
        HttpRequest {
            url: "https://example.googleapis.com/work",
            http_method: Post,
            headers: {
                "User-Agent": "Google-Cloud-Tasks",
            },
            body: [],
            authorization_header: Some(
                OauthToken(
                    OAuthToken {
                        service_account_email: "{service_account}",
                        scope: "scope",
                    },
                ),
            ),
        },
    ),
)
